// file: src/likertsurvey/LikertSurveyConcept.ts
import { Collection, Db } from "npm:mongodb";
import { Empty, ID } from "@utils/types.ts";
import { freshID } from "@utils/database.ts";

// Collection prefix to ensure namespace separation
const PREFIX = "LikertSurvey" + ".";

// Generic types for the concept's external dependencies
type Author = ID;
type Respondent = ID;

// Internal entity types, represented as IDs
type Survey = ID;
type Question = ID;

/**
 * State: A set of Surveys with an author, title, and scale.
 */
interface SurveyDoc {
  _id: Survey; // MongoDB document requires _id
  author: Author;
  title: string;
  scaleMin: number;
  scaleMax: number;
}

/**
 * State: A set of Questions, each linked to a survey and containing text.
 */
interface QuestionDoc {
  _id: Question; // MongoDB document requires _id
  survey: Survey;
  text: string;
}

/**
 * State: A set of Responses, linking a respondent, a question, and their chosen value.
 * Note: _id is automatically generated by freshID, but we don't treat Response as an explicit "ID type" in the concept spec.
 * The _id is internal for the collection, but the conceptual "entity" is the unique (respondent, question) pair.
 */
interface ResponseDoc {
  _id: ID; // An internal ID for the document itself
  respondent: Respondent;
  question: Question;
  value: number;
}

/**
 * @concept LikertSurvey [Author, Respondent]
 * @purpose To measure attitudes or opinions by asking respondents to rate their level of agreement with a series of statements on a predefined scale.
 * @principle If an author creates a survey with several questions on a defined scale (e.g., 1-5), and a respondent submits their answers within that scale, then the author can view the collected responses to analyze the respondent's opinions.
 */
export default class LikertSurveyConcept {
  surveys: Collection<SurveyDoc>;
  questions: Collection<QuestionDoc>;
  responses: Collection<ResponseDoc>;

  constructor(private readonly db: Db) {
    this.surveys = this.db.collection(PREFIX + "surveys");
    this.questions = this.db.collection(PREFIX + "questions");
    this.responses = this.db.collection(PREFIX + "responses");
  }

  /**
   * createSurvey (author: Author, title: String, scaleMin: Number, scaleMax: Number): (survey: Survey)
   *
   * @requires scaleMin < scaleMax
   *
   * @effects Creates a new survey with the given author, title, and scale; returns the ID of the new survey.
   */
  async createSurvey({ author, title, scaleMin, scaleMax }: { author: Author; title: string; scaleMin: number; scaleMax: number }): Promise<{ survey: Survey } | { error: string }> {
    if (scaleMin >= scaleMax) {
      return { error: "scaleMin must be strictly less than scaleMax" };
    }

    const surveyId = freshID() as Survey;
    await this.surveys.insertOne({ _id: surveyId, author, title, scaleMin, scaleMax });
    return { survey: surveyId };
  }

  /**
   * addQuestion (survey: Survey, text: String): (question: Question)
   *
   * @requires The survey must exist.
   *
   * @effects Adds a new question to the specified survey; returns the ID of the new question.
   */
  async addQuestion({ survey, text }: { survey: Survey; text: string }): Promise<{ question: Question } | { error: string }> {
    const existingSurvey = await this.surveys.findOne({ _id: survey });
    if (!existingSurvey) {
      return { error: `Survey with ID ${survey} not found.` };
    }

    const questionId = freshID() as Question;
    await this.questions.insertOne({ _id: questionId, survey, text });
    return { question: questionId };
  }

  /**
   * submitResponse (respondent: Respondent, question: Question, value: Number)
   *
   * @requires The question must exist.
   * @requires The respondent must not have already submitted a response for this question.
   * @requires The value must be within the survey's scale (inclusive of min and max).
   *
   * @effects Records the respondent's answer for the given question.
   */
  async submitResponse({ respondent, question, value }: { respondent: Respondent; question: Question; value: number }): Promise<Empty | { error: string }> {
    const questionDoc = await this.questions.findOne({ _id: question });
    if (!questionDoc) {
      return { error: `Question with ID ${question} not found.` };
    }

    const surveyDoc = await this.surveys.findOne({ _id: questionDoc.survey });
    if (!surveyDoc) {
      // This indicates a data integrity issue but is a good safeguard.
      return { error: "Associated survey for the question not found." };
    }

    if (value < surveyDoc.scaleMin || value > surveyDoc.scaleMax) {
      return { error: `Response value ${value} is outside the survey's scale [${surveyDoc.scaleMin}, ${surveyDoc.scaleMax}].` };
    }

    const existingResponse = await this.responses.findOne({ respondent, question });
    if (existingResponse) {
      return { error: "Respondent has already answered this question. Use updateResponse to change it." };
    }

    await this.responses.insertOne({ _id: freshID(), respondent, question, value }); // _id for the response document itself

    return {};
  }

  /**
   * updateResponse (respondent: Respondent, question: Question, value: Number)
   *
   * @requires The question must exist.
   * @requires The respondent must have already submitted a response for this question.
   * @requires The value must be within the survey's scale (inclusive of min and max).
   *
   * @effects Updates the respondent's existing answer for the given question.
   */
  async updateResponse({ respondent, question, value }: { respondent: Respondent; question: Question; value: number }): Promise<Empty | { error: string }> {
    const questionDoc = await this.questions.findOne({ _id: question });
    if (!questionDoc) {
      return { error: `Question with ID ${question} not found.` };
    }

    const surveyDoc = await this.surveys.findOne({ _id: questionDoc.survey });
    if (!surveyDoc) {
      return { error: "Associated survey for the question not found." };
    }

    if (value < surveyDoc.scaleMin || value > surveyDoc.scaleMax) {
      return { error: `Response value ${value} is outside the survey's scale [${surveyDoc.scaleMin}, ${surveyDoc.scaleMax}].` };
    }

    const result = await this.responses.updateOne({ respondent, question }, { $set: { value } });

    if (result.matchedCount === 0) {
      return { error: "No existing response found to update. Use submitResponse to create one." };
    }

    return {};
  }

  /**
   * _getSurvey(surveyId: Survey): (survey: { author: Author, title: String, scaleMin: Number, scaleMax: Number })
   *
   * @effects Returns an array containing the details of the specified survey (excluding its internal ID), or an empty array if not found.
   */
  async _getSurvey({ surveyId }: { surveyId: Survey }): Promise<{ survey: Omit<SurveyDoc, "_id"> }[]> {
    const surveyDoc = await this.surveys.findOne({ _id: surveyId });
    if (surveyDoc) {
      const { _id, ...details } = surveyDoc; // Omit _id as per spec's return type
      return [{ survey: details }];
    }
    return [];
  }

  /**
   * _getQuestionsForSurvey(survey: Survey): Question[]
   *
   * @effects Returns an array of Question IDs belonging to the specified survey.
   */
  async _getQuestionsForSurvey({ survey }: { survey: Survey }): Promise<Question[]> {
    const questions = await this.questions.find({ survey }).project({ _id: 1 }).toArray();
    return questions.map((q) => q._id);
  }

  /**
   * _getQuestionDetailsForSurvey(survey: Survey): (question: { id: Question, text: String })[]
   *
   * @effects Returns an array of question details (ID and text) for a given survey.
   */
  async _getQuestionDetailsForSurvey({ survey }: { survey: Survey }): Promise<{ question: { id: Question; text: string } }[]> {
    const questions = await this.questions.find({ survey }).project({ _id: 1, text: 1 }).toArray();
    return questions.map((q) => ({ question: { id: q._id, text: q.text } }));
  }

  /**
   * _getResponsesByQuestion(question: Question): (response: { respondent: Respondent, value: Number })[]
   *
   * @effects Returns an array of responses for a specific question, each including the respondent and their value.
   */
  async _getResponsesByQuestion({ question }: { question: Question }): Promise<{ response: { respondent: Respondent; value: number } }[]> {
    const responses = await this.responses.find({ question }).project({ respondent: 1, value: 1 }).toArray();
    return responses.map((r) => ({ response: { respondent: r.respondent, value: r.value } }));
  }

  /**
   * _getResponsesByRespondent(respondent: Respondent): (response: { question: Question, value: Number })[]
   *
   * @effects Returns an array of responses made by a specific respondent, each including the question and their value.
   */
  async _getResponsesByRespondent({ respondent }: { respondent: Respondent }): Promise<{ response: { question: Question; value: number } }[]> {
    const responses = await this.responses.find({ respondent }).project({ question: 1, value: 1 }).toArray();
    return responses.map((r) => ({ response: { question: r.question, value: r.value } }));
  }

  /**
   * _getSurveyResponseSummary(survey: Survey): (response: { question: Question, questionText: String, respondent: Respondent, value: Number })[]
   *
   * @effects Returns a comprehensive array of all responses for a given survey, including question ID, question text, respondent, and value.
   */
  async _getSurveyResponseSummary({ survey }: { survey: Survey }): Promise<{ response: { question: Question; questionText: string; respondent: Respondent; value: number } }[]> {
    const questionsForSurvey = await this.questions.find({ survey }).toArray();
    const questionIds = questionsForSurvey.map(q => q._id);

    if (questionIds.length === 0) {
      return []; // No questions in the survey, so no responses.
    }

    const responses = await this.responses.find({ question: { $in: questionIds } }).toArray();

    // Create a map for quick lookup of question text by question ID
    const questionMap = new Map<Question, string>();
    questionsForSurvey.forEach(q => questionMap.set(q._id, q.text));

    return responses.map(r => ({
      response: {
        question: r.question,
        questionText: questionMap.get(r.question) || "Unknown Question", // Fallback for safety
        respondent: r.respondent,
        value: r.value,
      },
    }));
  }
}
